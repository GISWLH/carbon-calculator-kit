核心逻辑：用能源活动数据 × 排放因子得到排放量，再按 Scope 1、Scope 2 拆分。

const factors = {
  electricity: 0.0005703, // tCO2e/kWh，演示系数，正式项目替换为适用因子
  gas: 0.002162,          // tCO2e/m3
  diesel: 0.00263         // tCO2e/L
};

const scope1 = gas * factors.gas + diesel * factors.diesel;
const grossScope2 = electricity * factors.electricity;
const greenOffset = grossScope2 * greenRate;
const scope2 = Math.max(grossScope2 - greenOffset, 0);
const total = scope1 + scope2;
const intensity = total / revenue;
